/**
 * Main webhook handler
 */
function doPost(e) {
  try {
    // Parse the incoming payload
    var payload = JSON.parse(e.postData.contents);
    
    // Verify secret key for security
    if (payload.secret !== SECRET_KEY) {
      return createResponse(false, 'Invalid secret key');
    }
    
    // Get the active spreadsheet
    var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
    
    // Route to appropriate handler based on type
    switch (payload.type) {
      case 'hiring':
        return handleHiringEnquiry(spreadsheet, payload.data);
      case 'talent':
        return handleTalentSubmission(spreadsheet, payload.data);
      case 'contact':
        return handleContact(spreadsheet, payload.data);
      default:
        return createResponse(false, 'Unknown submission type');
    }
    
  } catch (error) {
    Logger.log('Error in doPost: ' + error.toString());
    return createResponse(false, error.toString());
  }
}

/**
 * Handle hiring enquiry submissions
 */
function handleHiringEnquiry(spreadsheet, data) {
  try {
    var sheet = spreadsheet.getSheetByName('Hiring Enquiries');
    
    // Create sheet if it doesn't exist
    if (!sheet) {
      sheet = spreadsheet.insertSheet('Hiring Enquiries');
      // Add headers
      sheet.appendRow([
        'Timestamp',
        'Company',
        'Contact Name',
        'Email',
        'Phone',
        'Location',
        'Roles',
        'Seniority',
        'Timeline',
        'Details',
        'Contacted',
        'Contacted At',
        'Contacted By'
      ]);
      
      // Format header row
      var headerRange = sheet.getRange(1, 1, 1, 13);
      headerRange.setFontWeight('bold');
      headerRange.setBackground('#E91E63');
      headerRange.setFontColor('#FFFFFF');
      sheet.setFrozenRows(1);
    }
    
    // Format roles array for display
    var rolesText = '';
    if (data.roles && Array.isArray(data.roles)) {
      rolesText = data.roles.map(function(role) {
        return role.title + ' (x' + role.count + ')';
      }).join(', ');
    }
    
    // Append the new row
    sheet.appendRow([
      new Date(),
      data.company || '',
      data.contact_name || '',
      data.email || '',
      data.phone || '',
      data.location || '',
      rolesText,
      data.seniority || '',
      data.timeline || '',
      data.details || '',
      false,
      '',
      ''
    ]);
    
    // Auto-resize columns for better readability
    sheet.autoResizeColumns(1, 13);
    
    // Send automatic confirmation email to the employer
    if (SEND_AUTO_EMAILS) {
      try {
        sendHiringConfirmationEmail(data);
      } catch (emailError) {
        Logger.log('Email sending failed (non-critical): ' + emailError.toString());
        // Don't fail the webhook if email fails - data is already saved
      }
    }
    
    return createResponse(true, 'Hiring enquiry saved', sheet.getLastRow());
    
  } catch (error) {
    Logger.log('Error in handleHiringEnquiry: ' + error.toString());
    return createResponse(false, error.toString());
  }
}

/**

    // Create sheet if it doesn't exist
    if (!sheet) {
      sheet = spreadsheet.insertSheet('Talent Submissions');
      // Add headers
      sheet.appendRow([
        'Timestamp',
        'Name',
        'Email',
        'Phone',
        'Desired Role',
        'Custom Role',
        'Experience',
        'Location',
        'Links',
        'About',
        'CV URL',
        'CV File Name',
        'Contacted',
        'Contacted At',
        'Contacted By'
      ]);
      
      // Format header row
      var headerRange = sheet.getRange(1, 1, 1, 15);
      headerRange.setFontWeight('bold');
      headerRange.setBackground('#E91E63');
      headerRange.setFontColor('#FFFFFF');
      sheet.setFrozenRows(1);
    }
    
    // Determine the final role to display
    var finalRole = data.desired_role || '';
    if (data.desired_role === 'Other' && data.custom_role) {
      finalRole = data.custom_role;
    }
    
    // Append the new row
    sheet.appendRow([
      new Date(),
      data.name || '',
      data.email || '',
      data.phone || '',
      data.desired_role || '',
      data.custom_role || '',
      data.experience || '',
      data.location || '',
      data.links || '',
      data.about || '',
      data.cv_url || '',
      data.cv_file_name || '',
      false,
      '',
      ''
    ]);
    
    // Auto-resize columns for better readability
    sheet.autoResizeColumns(1, 15);

/**
 * Create a standardized JSON response
 */
function createResponse(success, message, id) {
  var response = {
    success: success,
    timestamp: new Date().toISOString()
  };
  
  if (success) {
    response.id = id || 'unknown';
    response.message = message || 'Success';
  } else {
    response.error = message || 'Unknown error';
  }
  
  return ContentService
    .createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

      }
